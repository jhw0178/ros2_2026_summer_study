### 3. open_manipulator_x에 카메라 추가

      <parent link="${prefix}link5"/>
      <child link="${prefix}camera_link"/>
      <origin xyz="0.07 0 0.05" rpy="0 0 0"/>
    </joint>

    <!-- camera_link 추가 -->

      <inertial>
