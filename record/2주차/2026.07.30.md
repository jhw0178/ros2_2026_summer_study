##### 1-1. 6개의 구역을 만들도록 벽 생성
    def add_wall(self):
        center_x = 0.0
        center_y = 0.0

        inner_radius = 0.12    # 중심부 빈 공간 확보 (반경 12cm까지는 벽 없음)
        wall_length = 0.25     # 벽의 길이
        wall_thickness = 0.02
        wall_height = 0.04     # 로봇이 넘나들 수 있는 낮은 높이

        for i in range(6):
            angle = math.radians(i * 60)

            collision_object = CollisionObject()
            collision_object.header.frame_id = "world"
            collision_object.id = f"wall_{i}"

            wall = SolidPrimitive()
            wall.type = SolidPrimitive.BOX
            wall.dimensions = [wall_length, wall_thickness, wall_height]

            pose = Pose()

            # 벽의 중심 좌표: 빈 공간(inner_radius) + 벽 길이의 절반
            offset = inner_radius + (wall_length / 2.0)
            pose.position.x = center_x + offset * math.cos(angle)
            pose.position.y = center_y + offset * math.sin(angle)
            pose.position.z = wall_height / 2.0

            # 벽 회전
            pose.orientation.z = math.sin(angle / 2.0)
            pose.orientation.w = math.cos(angle / 2.0)

            collision_object.primitives.append(wall)
            collision_object.primitive_poses.append(pose)
            collision_object.operation = CollisionObject.ADD

            self.planning_scene_monitor.process_collision_object(collision_object)

        self.get_logger().info("중앙이 빈 피자 조각 모양의 방사형 벽 6개 생성 완료")

### Gate B. OpenMANIPULATOR 소스와 빌드

    - 빌드 실패 패키지 0개
    - bringup, MoveIt config, GUI, teleop 패키지 또는 실행 파일 확인
    - 새 터미널에서도 두 setup 파일을 source한 뒤 같은 결과 확인

    - Humble 패키지나 과거 open_manipulator_x_* 패키지가 overlay에 섞이지 않았는지 확인한다.
    - 직접 설치가 교육생 PC별로 불안정하면 ROBOTIS 공식 Jazzy Docker 흐름을 대체 경로로 검증한다.

    - teleop과 GUI를 동시에 명령원으로 사용하지 않는다.
    - MoveIt start state, controller 이름, action server, simulation Play 상태를 확인한다.
### Gate F. WSL USB와 실기기
#### 입력 코드
#### Windows 관리자 Powershell
    $ usbipd list

    $ usbipd attach --wsl --busid <BUSID>
#### WSL
    $ lsusb
    $ ls -l /dev/ttyUSB* /dev/ttyACM*
    $ groups
    # 필요한 경우
    $ sudo usermod -aG dialout $USER
#### 실기기 실행 전 체크
    (1) 로봇 전원 OFF 상태에서 케이블과 관절 시작 자세 확인
    (2) 작업 반경에서 사람과 물체 제거
    (3) 비상 정지 또는 토크 해제 절차 확인
    (4) 포트가 U2D2인지 OpenCR인지 확인
    (5) 첫 동작은 한 관절, 낮은 속도, 작은 범위로 제한

    $ ros2 launch open_manipulator_bringup open_manipulator_x.launch.py

    $ ros2 topic echo /joint_states --once

    $ ros2 launch open_manipulator_moveit_config open_manipulator_x_moveit.launch.py

    - DYNAMIXEL 통신 오류 없이 bringup 완료
    - 실제 관절을 손으로 움직일 때 /joint_states가 일관되게 변함
    - 안전 Named State의 계획과 저속 실행 성공
    - gripper open/close 성공
##### 실패 조치
    - 자세 범위, 전원, 포트, 권한, USB attach, DYNAMIXEL ID와 firmware를 차례로 확인한다.
    - 실기기 문제를 해결하는 동안 수업은 Gate E 시뮬레이션으로 진행한다.
